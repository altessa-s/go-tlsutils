# Commit Scopes for go-tlsutils
# Use these scopes in conventional commits: type(scope): description
# Example: fix(ocsp): keep the cached staple on a soft failure

# Repository-wide scopes
deps                            # Dependency management
README                          # Root README
docs                            # Documentation

# Packages
tlsutils                        # Certificate loading and TLS defaults
ocsp                            # OCSP stapling, cache and refresh
providers                       # Certificate provider port
providers/file                  # PEM files on disk
providers/letsencrypt           # ACME / autocert
providers/s3                    # S3-backed PEM
providers/vault                 # Vault PKI
providers/internal/staplertest  # Test-only OCSP stapler double
spiffe                          # SPIFFE Workload API mTLS configs
internal/testhelpers            # Shared test fixtures
testdata                        # PEM fixtures

# Tooling
devtools                        # Development tool dependencies
ci                              # GitHub Actions workflows and scripts
